```python
import matplotlib.pyplot as plt
import numpy as np

# Example data (from XFOIL or literature)
cl = np.array([0.0, 0.5, 0.8, 1.0, 1.2, 1.4])

            label=f'Design Point (CL={cl[op_idx]}, L/D={cl[op_idx]/cd[op_idx]:.1f})', zorder=5)

# Stall region shading
```
